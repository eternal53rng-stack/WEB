const numbers = [4, 8, 15, 16, 23, 42];

let sum = 0;
let max = numbers[0]; 

let countMoreThan10 = 0;
for (let i = 0; i < numbers.length; i++) {
  if (numbers[i] > 10) {
    countMoreThan10 = countMoreThan10 + 1;
  }
}

const moreThan10 = [];
for (let i = 0; i < countMoreThan10; i++) {
  moreThan10[i] = 0;
}

let index = 0; 
for (let i = 0; i < numbers.length; i++) {
  let n = numbers[i];

  sum = sum + n;

  if (n > max) {
    max = n;
  }

  if (n > 10) {
    moreThan10[index] = n;
    index = index + 1;
  }
}

console.log('Сумма:', sum);                 
console.log('Максимальное число:', max);    
console.log('Числа больше 10:', moreThan10);
