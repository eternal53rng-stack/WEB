let students = [
  { name: 'Анна', grade: 85 },
  { name: 'Иван', grade: 60 },
  { name: 'Мария', grade: 92 },
  { name: 'Петр', grade: 74 }
];

let minGrade = 80;
let topStudents = [];
let sum = 0;

for (let i = 0; i < students.length; i++) {
  sum = sum + students[i].grade;

  if (students[i].grade > minGrade) {
    topStudents[topStudents.length] = students[i]; 
  }
}

let average = sum / students.length;

console.log('Средняя оценка:', average);
console.log('Хорошисты:', topStudents);
